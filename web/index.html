    📖 বাংলাভাষা সিনট্যাক্স চিটশিট (Syntax Quick Reference)
    
          
        
    
        
    
          
    
            
    
              
    ১. ডেটা টাইপ ও ভেরিয়েবল
    
              
    পূর্ণসংখ্যা x = 10;
দশমিক y = 3.14;
স্ট্রিং নাম = "বাংলাদেশ";
বুলিয়ান সত্য_কিনা = সত্য; // অথবা মিথ্যা
ধরি বয়স = 25; // টাইপ ইনফারেন্স
    

              
    ২. শর্ত ও ব্রাঞ্চিং (If - Else)
    
              
    যদি (x > 5 এবং সত্য_কিনা) {
    লেখো("ঠিক আছে");
} নাহলে {
    লেখো("ভুল");
}
    

              
    ৩. লুপ (While Loop)
    
              
    যতক্ষণ (x < 15) {
    লেখো(x);
    x = x + 1;
    যদি (x == 12) { থামো; } // break
}
    
            
    

            
    
              
    ৪. ফাংশন (Functions)
    
              
    ফাংশন যোগফল(পূর্ণসংখ্যা a, পূর্ণসংখ্যা b) -> পূর্ণসংখ্যা {
    ফেরত a + b;
}
লেখো(যোগফল(10, 20));
    

              
    ৫. স্ট্যাক ও কিউ (Stack & Queue)
    
              
    স্ট্যাক s;
s.ঠেলো(100);
লেখো(s.সামনে_দেখো()); // 100
লেখো(s.বের_করো());   // 100

কিউ q;
q.ঢোকাও("A");
লেখো(q.বের_করো_কিউ()); // A
    

              
    ৬. ক্লাস ও অবজেক্ট (OOP)
    
              
    ক্লাস ছাত্র {
    স্ট্রিং নাম = "অজানা";
    ফাংশন নাম_সেট(নিজে, স্ট্রিং নতুননাম) {
        নিজে.নাম = নতুননাম;
    }
}
সাকিব = নতুন ছাত্র();
সাকিব.নাম_সেট("সাকিব আল হাসান");
